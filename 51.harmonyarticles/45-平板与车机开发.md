# 45-平板与车机开发：大屏应用开发指南

> **本文目标**: 掌握平板和车机开发技术，包括分屏、悬浮窗、双栏布局、车机UI等大屏设备特有功能。

---

## 📖 引言

平板和车机是鸿蒙生态的重要组成部分，它们有独特的交互方式和使用场景：

```typescript
// ✅ 可运行代码
平板特性:
├── 大屏幕 (10-13寸)
├── 分屏多任务
├── 悬浮窗
├── 键盘鼠标支持
├── 手写笔输入
└── 横屏为主

车机特性:
├── 超宽屏 (16:9, 21:9)
├── 驾驶场景
├── 语音交互
├── 大字体/大按钮
├── 减少操作步骤
└── 夜间模式
```

### 为什么需要针对大屏优化？

```typescript
问题场景:
❌ 手机应用直接运行在平板 → 元素过小，浪费空间
❌ 不支持分屏 → 无法多任务处理
❌ 不适配车机 → 驾驶时操作困难，不安全

优化后:
✅ 充分利用大屏空间 → 双栏布局，一屏多内容
✅ 支持分屏多任务 → 提升生产力
✅ 适配车机场景 → 大按钮，语音交互，驾驶安全
```

#### 监听分屏状态

```typescript
// ✅ 可运行代码
/**
 * 监听窗口模式变化
 */
import window from '@ohos.window';

@Entry
@Component
struct SplitScreenExample {
  @State windowMode: string = 'fullscreen';
  @State windowWidth: number = 0;

  aboutToAppear() {
    this.listenWindowMode();
  }

  /**
   * 监听窗口模式变化
   */
  listenWindowMode() {
    window.getLastWindow(getContext(), (err, windowClass) => {
      if (err.code) {
        console.error('[SplitScreen] 获取窗口失败');
        return;
      }

      // 获取当前窗口模式
      const mode = windowClass.getWindowMode();
      this.updateWindowMode(mode);

      // 获取窗口尺寸
      const rect = windowClass.getWindowProperties().windowRect;
      this.windowWidth = rect.width;

      // 监听窗口模式变化
      windowClass.on('windowModeChange', (mode) => {
        console.log(`[SplitScreen] 窗口模式变化: ${mode}`);
        this.updateWindowMode(mode);
      });

      // 监听窗口尺寸变化
      windowClass.on('windowSizeChange', (size) => {
        console.log(`[SplitScreen] 窗口尺寸变化: ${size.width}x${size.height}`);
        this.windowWidth = size.width;
      });
    });
  }

  updateWindowMode(mode: number) {
    switch (mode) {
      case window.WindowMode.FULLSCREEN:
        this.windowMode = 'fullscreen';
        break;
      case window.WindowMode.SPLIT:
        this.windowMode = 'split';
        break;
      case window.WindowMode.FLOATING:
        this.windowMode = 'floating';
        break;
    }
  }

  build() {
    Column() {
      Text(`窗口模式: ${this.windowMode}`)
        .fontSize(20)
        .margin({ bottom: 20 })

      Text(`窗口宽度: ${this.windowWidth}px`)
        .fontSize(16)

      // 根据窗口模式调整布局
      if (this.windowMode === 'split') {
        this.buildCompactLayout()
      } else {
        this.buildFullLayout()
      }
    }
    .padding(20)
  }

  @Builder buildCompactLayout() {
    // 分屏模式: 紧凑布局
    Column({ space: 12 }) {
      Text('紧凑布局 (分屏)')
        .fontSize(18)

      List({ space: 8 }) {
        ForEach([1, 2, 3], (item) => {
          ListItem() {
            Text(`Item ${item}`)
              .width('100%')
              .height(60)
              .backgroundColor('#E0E0E0')
              .borderRadius(8)
              .textAlign(TextAlign.Center)
          }
        })
      }
    }
  }

  @Builder buildFullLayout() {
    // 全屏模式: 完整布局
    Column({ space: 16 }) {
      Text('完整布局 (全屏)')
        .fontSize(20)

      Grid() {
        ForEach([1, 2, 3, 4, 5, 6], (item) => {
          GridItem() {
            Column()
              .width('100%')
              .height(120)
              .backgroundColor('#E0E0E0')
              .borderRadius(12)
          }
        })
      }
      .columnsTemplate('1fr 1fr 1fr')
      .columnsGap(16)
      .rowsGap(16)
    }
  }
}
```

### 1.2 悬浮窗

#### 创建悬浮窗

```typescript
// ✅ 可运行代码
/**
 * 创建悬浮窗
 */
import window from '@ohos.window';

class FloatingWindowManager {
  private floatingWindow: window.Window | null = null;

  /**
   * 创建悬浮窗
   */
  async createFloatingWindow(url: string): Promise<void> {
    try {
      // 创建子窗口
      this.floatingWindow = await window.createWindow({
        name: 'floatingWindow',
        windowType: window.WindowType.TYPE_FLOAT,
        ctx: getContext()
      });

      // 设置悬浮窗属性
      await this.floatingWindow.resize(400, 300);
      await this.floatingWindow.moveWindowTo(100, 100);

      // 设置窗口模式
      await this.floatingWindow.setWindowMode(window.WindowMode.FLOATING);

      // 加载内容
      await this.floatingWindow.setUIContent(url);

      // 显示窗口
      await this.floatingWindow.showWindow();

      console.log('[FloatingWindow] 悬浮窗创建成功');
    } catch (error) {
      console.error('[FloatingWindow] 创建失败:', error);
    }
  }

  /**
   * 关闭悬浮窗
   */
  async closeFloatingWindow(): Promise<void> {
    if (this.floatingWindow) {
      await this.floatingWindow.destroyWindow();
      this.floatingWindow = null;
      console.log('[FloatingWindow] 悬浮窗已关闭');
    }
  }

  /**
   * 移动悬浮窗
   */
  async moveWindow(x: number, y: number): Promise<void> {
    if (this.floatingWindow) {
      await this.floatingWindow.moveWindowTo(x, y);
    }
  }

  /**
   * 调整悬浮窗大小
   */
  async resizeWindow(width: number, height: number): Promise<void> {
    if (this.floatingWindow) {
      await this.floatingWindow.resize(width, height);
    }
  }
}

// 全局单例
export const floatingWindowManager = new FloatingWindowManager();

// 使用示例
@Entry
@Component
struct MainPage {
  build() {
    Column() {
      Button('打开悬浮窗')
        .onClick(() => {
          floatingWindowManager.createFloatingWindow('pages/FloatingPage');
        })

      Button('关闭悬浮窗')
        .onClick(() => {
          floatingWindowManager.closeFloatingWindow();
        })
    }
  }
}
```

## 二、双栏布局 (Master-Detail)

### 2.1 双栏布局原理

```typescript
// ✅ 可运行代码
Master-Detail布局:
┌─────────────────────────────────────┐
│ Master (主列表)  │  Detail (详情)   │
│                 │                   │
│ ┌─────────────┐ │ ┌───────────────┐│
│ │ 项目1       │ │ │               ││
│ ├─────────────┤ │ │               ││
│ │ 项目2  ✓    │ │ │   详情内容     ││
│ ├─────────────┤ │ │               ││
│ │ 项目3       │ │ │               ││
│ └─────────────┘ │ └───────────────┘│
│                 │                   │
│   30%-40%       │    60%-70%        │
└─────────────────────────────────────┘

适用场景:
- 邮件应用 (邮件列表 + 邮件详情)
- 设置应用 (设置项 + 设置详情)
- 新闻应用 (新闻列表 + 新闻详情)
- 文件管理 (文件列表 + 文件详情)
```

```typescript
@Entry
@Component
struct MasterDetailLayout {
  @State selectedItem: NewsItem | null = null;
  @State newsList: NewsItem[] = [];

  build() {
    Row() {
      // Master: 列表区 (左侧)
      Column() {
        Text('新闻列表')
          .fontSize(20)
          .fontWeight(FontWeight.Bold)
          .padding({ top: 24, bottom: 16, left: 24, right: 24 })

        List({ space: 8 }) {
          ForEach(this.newsList, (news: NewsItem) => {
            ListItem() {
              this.buildMasterItem(news)
            }
            .onClick(() => {
              this.selectedItem = news;
            })
          })
        }
        .layoutWeight(1)
        .padding({ left: 16, right: 16 })
      }
      .width('35%')
      .height('100%')
      .backgroundColor('#F5F5F5')

      // 分割线
      Column()
        .width(1)
        .height('100%')
        .backgroundColor('#E0E0E0')

      // Detail: 详情区 (右侧)
      Column() {
        if (this.selectedItem) {
          this.buildDetailContent(this.selectedItem)
        } else {
          this.buildEmptyState()
        }
      }
      .layoutWeight(1)
      .height('100%')
      .backgroundColor(Color.White)
    }
    .width('100%')
    .height('100%')
  }

  /**
   * Master项
   */
  @Builder buildMasterItem(news: NewsItem) {
    Row() {
      // 缩略图
      Image(news.thumbnail)
        .width(80)
        .height(60)
        .borderRadius(8)
        .objectFit(ImageFit.Cover)

      // 标题和摘要
      Column({ space: 8 }) {
        Text(news.title)
          .fontSize(16)
          .fontWeight(FontWeight.Bold)
          .maxLines(2)
          .textOverflow({ overflow: TextOverflow.Ellipsis })

        Text(news.summary)
          .fontSize(14)
          .fontColor('#666666')
          .maxLines(1)
          .textOverflow({ overflow: TextOverflow.Ellipsis })
      }
      .layoutWeight(1)
      .margin({ left: 12 })
      .alignItems(HorizontalAlign.Start)
    }
    .width('100%')
    .padding(12)
    .backgroundColor(this.selectedItem?.id === news.id ? '#E6F2FF' : Color.White)
    .borderRadius(12)
  }

  /**
   * 详情内容
   */
  @Builder buildDetailContent(news: NewsItem) {
    Scroll() {
      Column({ space: 24 }) {
        // 标题
        Text(news.title)
          .fontSize(28)
          .fontWeight(FontWeight.Bold)

        // 元信息
        Row({ space: 16 }) {
          Text(news.author)
            .fontSize(14)
            .fontColor('#666666')

          Text(news.publishTime)
            .fontSize(14)
            .fontColor('#999999')

          Text(`${news.views}阅读`)
            .fontSize(14)
            .fontColor('#999999')
        }

        // 封面图
        Image(news.coverImage)
          .width('100%')
          .borderRadius(12)

        // 正文
        Text(news.content)
          .fontSize(16)
          .lineHeight(28)

        // 相关新闻
        Column({ space: 12 }) {
          Text('相关新闻')
            .fontSize(18)
            .fontWeight(FontWeight.Bold)

          ForEach(news.relatedNews, (related: NewsItem) => {
            Row() {
              Image(related.thumbnail)
                .width(60)
                .height(45)
                .borderRadius(6)

              Text(related.title)
                .fontSize(14)
                .maxLines(2)
                .layoutWeight(1)
                .margin({ left: 12 })
            }
            .width('100%')
            .padding(12)

          })
        }
        .alignItems(HorizontalAlign.Start)
      }

      .alignItems(HorizontalAlign.Start)
    }
  }

  /**
   * 空状态
   */
  @Builder buildEmptyState() {
    Column() {
      Text('选择一条新闻查看详情')
        .fontSize(16)
        .fontColor('#999999')
    }
    .width('100%')
    .height('100%')
    .justifyContent(FlexAlign.Center)
  }

  aboutToAppear() {

    this.loadNewsList();
  }

        summary: '华为发布鸿蒙4.0系统...',
        thumbnail: 'https://example.com/thumb1.jpg',
        // 更多字段...
      },
      // 更多新闻...
    ];

    // 默认选中第一条
    if (this.newsList.length > 0) {
      this.selectedItem = this.newsList[0];
    }
  }
```

```typescript
/**
 * 响应式双栏布局
 * - 平板: 双栏
 * - 手机: 单栏 (列表/详情切换)
 */
@Entry
@Component
struct ResponsiveMasterDetail {
  @State currentBreakpoint: Breakpoint = Breakpoint.SM;
  @State selectedItem: Item | null = null;
  @State showDetail: boolean = false; // 手机端是否显示详情

  aboutToAppear() {
    breakpointManager.init();
    this.currentBreakpoint = breakpointManager.getCurrentBreakpoint();

    breakpointManager.onChange((breakpoint) => {
      this.currentBreakpoint = breakpoint;
    });
  }

  build() {
    if (this.currentBreakpoint >= Breakpoint.MD) {
      // 平板: 双栏布局
      this.buildTabletLayout()
    } else {
      // 手机: 单栏布局
      this.buildPhoneLayout()
    }
  }

  @Builder buildTabletLayout() {
    Row() {
      // Master
      this.buildMaster()

      // Detail
      this.buildDetail()
    }
  }

  @Builder buildPhoneLayout() {
    if (this.showDetail && this.selectedItem) {
      // 显示详情
      Column() {
        // 返回按钮
        Row() {
          Button('< 返回')
            .onClick(() => {
              this.showDetail = false;
            })
        }
        .width('100%')
        .height(56)
        .padding({ left: 16, right: 16 })

        // 详情内容
        this.buildDetail()
      }
    } else {
      // 显示列表
      this.buildMaster()
    }
  }

  @Builder buildMaster() {
    // Master内容
  }

  @Builder buildDetail() {
    // Detail内容
  }

  onItemClick(item: Item) {
    this.selectedItem = item;

    // 手机端: 切换到详情页
    if (this.currentBreakpoint < Breakpoint.MD) {
      this.showDetail = true;
    }
  }
}
```

```typescript
/**
 * 语音交互管理器

import speechRecognizer from '@ohos.ai.intelligentVoice.speechRecognizer';

class VoiceManager {
  private recognizer: speechRecognizer.SpeechRecognizer | null = null;

  /**
   * 初始化语音识别
   */
  async init(): Promise<void> {
    try {
      this.recognizer = await speechRecognizer.createSpeechRecognizer();
      console.log('[Voice] 初始化成功');
    } catch (error) {
      console.error('[Voice] 初始化失败:', error);
    }
  }

  /**
   * 开始语音识别
   */
  async startListening(callback: (text: string) => void): Promise<void> {
    if (!this.recognizer) {
      console.error('[Voice] 未初始化');
      return;
    }

    try {
      // 设置参数
      await this.recognizer.setParameter({
        language: 'zh-CN',
        continuous: false  // 单次识别
      });

      // 开始识别
      await this.recognizer.start();
      console.log('[Voice] 开始识别');

      // 监听识别结果
      this.recognizer.on('result', (result) => {
        const text = result.result[0];
        console.log(`[Voice] 识别结果: ${text}`);
        callback(text);
      });

      // 监听错误
      this.recognizer.on('error', (error) => {
        console.error('[Voice] 识别错误:', error);
      });
    } catch (error) {
      console.error('[Voice] 启动失败:', error);
    }
  }

  /**
   * 停止语音识别
   */
  async stopListening(): Promise<void> {
    if (this.recognizer) {
      await this.recognizer.stop();
      console.log('[Voice] 停止识别');
    }
  }

  /**
   * 释放资源
   */
  async release(): Promise<void> {
    if (this.recognizer) {
      await this.recognizer.release();
      this.recognizer = null;
      console.log('[Voice] 释放资源');
    }
  }
}
```
