## 引言：为什么需要系统化的状态管理？

想象你在开发一个电商应用：

```typescript
// ✅ 可运行代码
// 😓 没有状态管理的混乱代码
@Component
struct ProductPage {
  @State cart: Product[] = []          // 购物车数据
  @State userInfo: User = new User()   // 用户信息
  @State theme: string = 'light'        // 主题

  build() {
    Column() {
      // 需要把数据传递给10+个子组件...
      Header({ cart: $cart, user: $userInfo, theme: $theme })
      ProductList({ cart: $cart, theme: $theme })
      Footer({ cart: $cart, user: $userInfo })
    }
  }
}
```

**痛点**:
- ❌ **属性透传地狱**: 数据需要一层层传递
- ❌ **状态分散**: 购物车、用户信息散落各处
- ❌ **难以调试**: 不知道谁修改了数据
- ❌ **性能问题**: 任何状态变化都重新渲染整个页面

**本文将带你系统化学习ArkTS状态管理**, 解决以上所有问题！

## 1. 状态管理核心概念

### 1.1 什么是状态管理？

**状态管理** = **数据存储** + **变化追踪** + **UI同步**

```typescript
// ✅ 可运行代码
// 1. 数据存储
@State count: number = 0

// 2. 变化追踪（装饰器自动完成）
this.count++  // 自动记录变化

// 3. UI同步（自动重新渲染）
Text(`${this.count}`)  // 自动更新
```

### 1.2 状态分类

| 类型 | 说明 | 装饰器 | 示例 |
|------|------|--------|------|
| **组件状态** | 单个组件私有 | `@State` | 表单输入值、展开状态 |
| **共享状态** | 多组件共享 | `@Link`, `@Provide` | 购物车、用户信息 |
| **全局状态** | 跨页面共享 | `AppStorage` | 主题、语言设置 |
| **持久状态** | 需要保存到本地 | `PersistentStorage` | 登录状态、偏好设置 |

### 2.2 @State - 组件私有状态

**适用场景**: 只在当前组件内使用的状态

```typescript
// ✅ 可运行代码
@Component
struct ExpandablePanel {
  @State isExpanded: boolean = false  // 展开状态

  build() {
    Column() {
      // 标题栏
      Row()
        .onClick(() => {
          this.isExpanded = !this.isExpanded  // ✅ 只在此组件内修改
        })

      // 内容区（根据状态显示/隐藏）
      if (this.isExpanded) {
        Text('详细内容...')
      }
    }
  }
}
```

**特性**:
- ✅ **简单直接**: 不需要父组件参与
- ✅ **性能好**: 状态变化只影响当前组件
- ⚠️ **局限性**: 无法与其他组件共享

### 2.2 @Prop - 单向数据流

**适用场景**: 子组件需要展示父组件的数据，但不修改

```typescript
// ✅ 可运行代码
// 父组件
@Component
struct ProductList {
  @State products: Product[] = getProducts()

  build() {
    List() {
      ForEach(this.products, (product: Product) => {
        ListItem() {
          ProductCard({ product: product })  // 传递数据
        }
      })
    }
  }
}

// 子组件
@Component
struct ProductCard {
  @Prop product: Product  // 只读属性

  build() {
    Column() {
      Image(this.product.image)
      Text(this.product.name)
      Text(`¥${this.product.price}`)
    }
  }
}
```

**数据流向**:
```typescript
// ✅ 可运行代码
父组件 @State → 子组件 @Prop
   ↓                 ↓
 可修改            只读取
```

**优点**:
- ✅ **数据流清晰**: 只能父→子，不会反向污染
- ✅ **易于调试**: 数据来源明确
- ✅ **类型安全**: 编译时检查

### 2.3 @Link - 双向绑定

**适用场景**: 父子组件需要同步修改同一份数据

```typescript
// ✅ 可运行代码
// 父组件
@Component
struct ShoppingCart {
  @State selectedItems: Set<number> = new Set()

  build() {
    Column() {
      // 全选按钮
      SelectAllButton({ selectedItems: $selectedItems })

      // 商品列表
      List() {
        ForEach(this.items, (item: CartItem) => {
          CartItemRow({
            item: item,
            selectedItems: $selectedItems  // ✅ 使用$传递引用
          })
        })
      }
    }
  }
}

// 子组件
@Component
struct SelectAllButton {
  @Link selectedItems: Set<number>  // 可修改

  build() {
    Button('全选')
      .onClick(() => {
        // ✅ 修改会同步到父组件
        this.selectedItems = new Set(this.allItemIds)
      })
  }
}

@Component
struct CartItemRow {
  item: CartItem
  @Link selectedItems: Set<number>

  build() {
    Row() {
      Checkbox()
        .select(this.selectedItems.has(this.item.id))
        .onChange(checked => {
          // ✅ 修改会同步到父组件和兄弟组件
          if (checked) {
            this.selectedItems.add(this.item.id)
          } else {
            this.selectedItems.delete(this.item.id)
          }
          // 触发更新
          this.selectedItems = new Set(this.selectedItems)
        })

      Text(this.item.name)
    }
  }
}
```

**数据流向**:
```typescript
// ✅ 可运行代码
父组件 @State ←→ 子组件 @Link
      ↕              ↕
   双向同步       双向同步
```

**关键点**:
- 传递时必须使用`$`符号: `{ data: $data }`
- 父子组件共享同一份数据引用
- 任意一方修改都会同步

### 2.4 @Observed + @ObjectLink - 嵌套对象响应式

**问题**: `@State`只监听对象引用变化

```typescript
@State user: User = new User()

// ❌ 不会触发UI更新（引用未变）
this.user.name = '新名字'

// ✅ 必须替换整个对象
this.user = { ...this.user, name: '新名字' }
```

**解决方案**: 使用`@Observed`深度监听

```typescript
// ✅ 可运行代码
// 1. 标记类为可观察
@Observed
class User {
  name: string = ''
  age: number = 0
  address: Address = new Address()
}

@Observed
class Address {
  city: string = ''
  street: string = ''
}

// 2. 父组件持有数据
@Component
struct UserProfile {
  @State user: User = new User()

  build() {
    Column() {
      // 使用@ObjectLink传递
      UserForm({ user: this.user })
      AddressForm({ address: this.user.address })
    }
  }
}

// 3. 子组件使用@ObjectLink
@Component
struct UserForm {
  @ObjectLink user: User  // 深度监听

  build() {
    Column() {
      TextInput({ text: this.user.name })
        .onChange(value => {
          // ✅ 直接修改属性，自动触发UI更新
          this.user.name = value
        })

      TextInput({ text: this.user.age.toString() })
        .onChange(value => {
          this.user.age = parseInt(value)
        })
    }
  }
}

@Component
struct AddressForm {
  @ObjectLink address: Address

  build() {
    Column() {
      TextInput({ text: this.address.city })
        .onChange(value => {
          this.address.city = value  // ✅ 深层对象也能监听
        })
    }
  }
}
```

**对比总结**:

| 场景 | 使用方案 | 代码 |
|------|----------|------|
| 简单类型 | `@State` | `@State count: number = 0` |
| 对象引用 | `@State` + 替换对象 | `this.user = {...user}` |
| 对象属性 | `@Observed` + `@ObjectLink` | `this.user.name = 'xx'` |

## 3. 跨组件状态共享

### 3.1 @Provide + @Consume - 依赖注入

**场景**: 跨多层级传递数据，避免属性透传

**传统方式（属性透传地狱）**:
```typescript
// ✅ 可运行代码
// 😓 需要一层层传递
@Component
struct App {
  @State theme: string = 'dark'

  build() {
    PageA({ theme: this.theme })
  }
}

@Component
struct PageA {
  @Prop theme: string  // 被迫接收，但自己不用

  build() {
    PageB({ theme: this.theme })  // 继续传递
  }
}

@Component
struct PageB {
  @Prop theme: string  // 被迫接收，但自己不用

  build() {
    ThemedButton({ theme: this.theme })  // 继续传递
  }
}

@Component
struct ThemedButton {
  @Prop theme: string  // 终于用到了！

  build() {
    Button().backgroundColor(this.theme === 'dark' ? '#333' : '#fff')
  }
}
```

**使用@Provide/@Consume**:
```typescript
// ✅ 可运行代码
// ✅ 祖先组件提供数据
@Component
struct App {
  @Provide theme: string = 'dark'  // 提供数据

  build() {
    PageA()  // 不需要传递
  }
}

// 中间组件不需要处理theme
@Component
struct PageA {
  build() {
    PageB()  // 不需要传递
  }
}

@Component
struct PageB {
  build() {
    ThemedButton()  // 不需要传递
  }
}

// 深层组件直接消费
@Component
struct ThemedButton {
  @Consume theme: string  // 直接获取祖先提供的数据

  build() {
    Button()
      .backgroundColor(this.theme === 'dark' ? '#333' : '#fff')
      .onClick(() => {
        // ✅ 可以修改，会同步到所有使用者
        this.theme = this.theme === 'dark' ? 'light' : 'dark'
      })
  }
}
```

**工作原理**:
```typescript
// ✅ 可运行代码
App (@Provide theme)
  ↓
PageA (跳过)
  ↓
PageB (跳过)
  ↓
ThemedButton (@Consume theme) ✅ 直接获取
```

**特性**:
- ✅ **跨层级传递**: 无需中间组件参与
- ✅ **双向绑定**: 深层组件可修改数据
- ✅ **自动同步**: 所有消费者同步更新
- ⚠️ **命名匹配**: `@Provide`和`@Consume`的变量名必须一致

**指定别名**:
```typescript
// ✅ 可运行代码
// 提供时指定key
@Provide('appTheme') myTheme: string = 'dark'

// 消费时使用相同key
@Consume('appTheme') theme: string
```

### 3.2 实战案例：主题切换系统

```typescript
// ✅ 可运行代码
// 主题数据类
@Observed
class ThemeConfig {
  mode: 'light' | 'dark' = 'light'
  primaryColor: string = '#1890ff'
  fontSize: number = 14

  get backgroundColor(): string {
    return this.mode === 'dark' ? '#1a1a1a' : '#ffffff'
  }

  get textColor(): string {
    return this.mode === 'dark' ? '#ffffff' : '#333333'
  }
}

// 根组件
@Entry
@Component
struct App {
  @Provide('theme') themeConfig: ThemeConfig = new ThemeConfig()

  build() {
    Column() {
      HomePage()
    }
    .backgroundColor(this.themeConfig.backgroundColor)
  }
}

// 首页
@Component
struct HomePage {
  build() {
    Column() {
      Header()
      Content()
      ThemeSettings()  // 设置页面，可能嵌套很深
    }
  }
}

// 深层组件：主题设置
@Component
struct ThemeSettings {
  @Consume('theme') themeConfig: ThemeConfig  // 直接获取

  build() {
    Column() {
      Text('主题设置')
        .fontSize(20)
        .fontColor(this.themeConfig.textColor)

      // 切换深色模式
      Row() {
        Text('深色模式')
        Toggle({ type: ToggleType.Switch, isOn: this.themeConfig.mode === 'dark' })
          .onChange(isOn => {
            // ✅ 修改后，整个应用主题立即切换
            this.themeConfig.mode = isOn ? 'dark' : 'light'
          })
      }

      // 调整字体大小
      Row() {
        Text('字体大小')
        Slider({ value: this.themeConfig.fontSize, min: 12, max: 20 })
          .onChange(value => {
            this.themeConfig.fontSize = value
          })
      }
    }
  }
}

// 其他任何组件都可以直接消费主题
@Component
struct SomeDeepComponent {
  @Consume('theme') themeConfig: ThemeConfig

  build() {
    Text('我会自动响应主题变化')
      .fontSize(this.themeConfig.fontSize)
      .fontColor(this.themeConfig.textColor)
  }
}
```

## 4. 全局状态管理

### 4.1 AppStorage - 应用级全局状态

**特点**: 单例存储，整个应用共享

```typescript
// ✅ 可运行代码
// 1. 初始化全局状态（通常在入口文件）
AppStorage.SetOrCreate('userInfo', {
  name: '游客',
  isLogin: false
})

AppStorage.SetOrCreate('cartCount', 0)

// 2. 在组件中使用 - @StorageProp（单向）
@Component
struct Header {
  @StorageProp('userInfo') userInfo: UserInfo = { name: '', isLogin: false }

  build() {
    Row() {
      Text(this.userInfo.name)
      if (!this.userInfo.isLogin) {
        Button('登录')
      }
    }
  }
}

// 3. 在组件中使用 - @StorageLink（双向）
@Component
struct CartIcon {
  @StorageLink('cartCount') cartCount: number = 0  // 双向同步

  build() {
    Badge({ count: this.cartCount }) {
      Image($r('app.media.cart'))
    }
    .onClick(() => {
      // ✅ 修改会同步到AppStorage和所有使用者
      this.cartCount++
    })
  }
}

// 4. 在普通函数中使用
function addToCart(product: Product) {
  let count = AppStorage.Get<number>('cartCount') || 0
  AppStorage.Set('cartCount', count + 1)  // 所有@StorageLink自动更新
}
```

**@StorageProp vs @StorageLink**:

| 装饰器 | 数据流 | 修改行为 | 使用场景 |
|--------|--------|----------|----------|
| `@StorageProp` | AppStorage → 组件 | 组件修改不影响全局 | 只读展示 |
| `@StorageLink` | AppStorage ↔ 组件 | 组件修改同步到全局 | 可修改数据 |

### 4.2 LocalStorage - 页面级状态

**特点**: 独立的存储实例，多个页面可拥有各自的LocalStorage

```typescript
// ✅ 可运行代码
// 创建LocalStorage实例
let storage = new LocalStorage({ count: 0 })

// 页面入口绑定storage
@Entry(storage)
@Component
struct PageA {
  build() {
    Column() {
      ComponentA()
      ComponentB()
    }
  }
}

// 组件使用
@Component
struct ComponentA {
  @LocalStorageProp('count') count: number = 0  // 单向

  build() {
    Text(`Count: ${this.count}`)
  }
}

@Component
struct ComponentB {
  @LocalStorageLink('count') count: number = 0  // 双向

  build() {
    Button('加1')
      .onClick(() => this.count++)  // 同页面的ComponentA会更新
  }
}
```

**AppStorage vs LocalStorage**:

| 特性 | AppStorage | LocalStorage |
|------|------------|--------------|
| 作用域 | 全应用共享 | 单页面共享 |
| 生命周期 | 应用级 | 页面级 |
| 适用场景 | 用户信息、全局配置 | 页面临时状态 |

### 4.3 PersistentStorage - 持久化存储

**特点**: 数据自动保存到本地，应用重启后恢复

```typescript
// ✅ 可运行代码
// 1. 配置持久化（应用启动时）
PersistentStorage.PersistProp('token', '')
PersistentStorage.PersistProp('userPreferences', {
  theme: 'light',
  language: 'zh-CN'
})

// 2. 使用（与AppStorage完全一致）
@Component
struct LoginPage {
  @StorageLink('token') token: string = ''

  build() {
    Column() {
      if (this.token) {
        Text('已登录')
      } else {
        Button('登录')
          .onClick(async () => {
            const token = await login()
            this.token = token  // ✅ 自动持久化到本地
          })
      }
    }
  }
}

// 3. 应用重启后自动恢复
// PersistentStorage会从本地读取token，自动同步到AppStorage
```

**持久化机制**:
```typescript
// ✅ 可运行代码
组件修改 → AppStorage → PersistentStorage → 本地存储
                                             ↓
应用重启 ← AppStorage ← PersistentStorage ← 读取本地
```

## 5. 状态管理最佳实践

### 5.1 状态设计原则

**原则1：最小化状态**

```typescript
// ❌ 冗余状态
@State firstName: string = '张'
@State lastName: string = '三'
@State fullName: string = '张三'  // 冗余！可通过计算得出

// ✅ 最小化状态
@State firstName: string = '张'
@State lastName: string = '三'
get fullName(): string {  // 计算属性
  return this.firstName + this.lastName
}
```

**原则2：单一数据源**

```typescript

```

**原则3：状态归一化**

  author: { id: number, name: string }

## 6. 实战案例：电商购物车

### 6.1 需求分析

**功能**:
- 添加/删除商品
- 修改数量
- 全选/反选
- 计算总价
- 持久化（刷新不丢失）

### 6.2 状态设计

```typescript
// ✅ 可运行代码
// 数据模型
@Observed
class CartItem {
  id: number
  name: string
  price: number
  quantity: number
  selected: boolean
  image: string

  constructor(id: number, name: string, price: number, image: string) {
    this.id = id
    this.name = name
    this.price = price
    this.quantity = 1
    this.selected = true
    this.image = image
  }

  get subtotal(): number {
    return this.price * this.quantity
  }
}

// 购物车Store
@Observed
class CartStore {
  items: CartItem[] = []

  // 添加商品
  addItem(product: Product) {
    const existing = this.items.find(item => item.id === product.id)
    if (existing) {
      existing.quantity++
    } else {
      this.items.push(new CartItem(
        product.id,
        product.name,
        product.price,
        product.image
      ))
    }
    this.save()  // 持久化
  }

  // 删除商品
  removeItem(id: number) {
    const index = this.items.findIndex(item => item.id === id)
    if (index > -1) {
      this.items.splice(index, 1)
      this.save()
    }
  }

  // 修改数量
  updateQuantity(id: number, quantity: number) {
    const item = this.items.find(item => item.id === id)
    if (item) {
      item.quantity = Math.max(1, quantity)
      this.save()
    }
  }

  // 切换选中
  toggleSelect(id: number) {
    const item = this.items.find(item => item.id === id)
    if (item) {
      item.selected = !item.selected
      this.save()
    }
  }

  // 全选/反选
  toggleSelectAll() {
    const allSelected = this.items.every(item => item.selected)
    this.items.forEach(item => item.selected = !allSelected)
    this.save()
  }

  // 计算总价
  get totalPrice(): number {
    return this.items
      .filter(item => item.selected)
      .reduce((sum, item) => sum + item.subtotal, 0)
  }

  // 选中数量
  get selectedCount(): number {
    return this.items.filter(item => item.selected).length
  }

  // 是否全选
  get isAllSelected(): boolean {
    return this.items.length > 0 && this.items.every(item => item.selected)
  }

  // 持久化
  save() {
    AppStorage.Set('cart', JSON.stringify(this.items))
  }

  // 恢复数据
  static restore(): CartStore {
    const store = new CartStore()
    const data = AppStorage.Get<string>('cart')
    if (data) {
      try {
        store.items = JSON.parse(data).map((item: any) =>
          Object.assign(new CartItem(0, '', 0, ''), item)
        )
      } catch (e) {
        console.error('恢复购物车失败', e)
      }
    }
    return store
  }
}
```

## 总结

### 核心要点

1. **装饰器体系**:
   - `@State/@Prop/@Link` - 组件级
   - `@Provide/@Consume` - 跨层级
   - `@StorageLink/@StorageProp` - 全局级

2. **嵌套对象**:
   - 使用`@Observed`标记类
   - 使用`@ObjectLink`传递

3. **全局状态**:
   - `AppStorage` - 应用级共享
   - `LocalStorage` - 页面级共享
   - `PersistentStorage` - 持久化

4. **性能优化**:
   - 最小化状态
   - 拆分组件
   - 不可变更新
