export class CurrentWeather {
  city: string = ''              // 城市名
  temperature: number = 0        // 温度
  weather: string = ''           // 天气状况
  weatherIcon: string = ''       // 天气图标
  humidity: number = 0           // 湿度
  windSpeed: number = 0          // 风速
  windDirection: string = ''     // 风向
  updateTime: string = ''        // 更新时间

  constructor(data?: Partial<CurrentWeather>) {
    if (data) {
      Object.assign(this, data)
    }
  }
}

// 每日天气预报
export class DailyForecast {
  date: string = ''              // 日期
  weekday: string = ''           // 星期
  tempMax: number = 0            // 最高温
  tempMin: number = 0            // 最低温
  weather: string = ''           // 天气
  weatherIcon: string = ''       // 图标

  constructor(data?: Partial<DailyForecast>) {
    if (data) {
      Object.assign(this, data)
    }
  }
}

// 完整天气数据
export class WeatherData {
  current: CurrentWeather = new CurrentWeather()
  forecast: DailyForecast[] = []

  constructor(data?: Partial<WeatherData>) {
    if (data) {
      if (data.current) {
        this.current = new CurrentWeather(data.current)
      }
      if (data.forecast) {
        this.forecast = data.forecast.map(item => new DailyForecast(item))
      }
    }
  }
}

// ✅ 可运行代码
import http from '@ohos.net.http'
import hilog from '@ohos.hilog'

export class HttpUtil {
  private static readonly TAG = 'HttpUtil'

  /**
   * GET请求
   */
  static async get(url: string): Promise<string> {
    return new Promise((resolve, reject) => {
      let httpRequest = http.createHttp()

      httpRequest.request(url, {
        method: http.RequestMethod.GET,
        connectTimeout: 10000,
        readTimeout: 10000,
        header: {
          'Content-Type': 'application/json'
        }
      }).then((response) => {
        hilog.info(0x0000, this.TAG, `Response: ${response.responseCode}`)

        if (response.responseCode === 200) {
          resolve(response.result as string)
        } else {
          reject(new Error(`HTTP Error: ${response.responseCode}`))
        }

        httpRequest.destroy()
      }).catch((err: Error) => {
        hilog.error(0x0000, this.TAG, `Request failed: ${err.message}`)
        reject(err)
        httpRequest.destroy()
      })
    })
  }
}

### 3.2 天气API服务

**创建**: `service/WeatherService.ets`

```typescript
// ✅ 可运行代码
import { HttpUtil } from '../common/utils/HttpUtil'
import { WeatherData, CurrentWeather, DailyForecast } from '../model/WeatherModel'

export class WeatherService {
  // 使用和风天气API (免费版)
  private static readonly API_KEY = 'YOUR_API_KEY'  // 替换为你的KEY
  private static readonly BASE_URL = 'https://devapi.qweather.com/v7'

  /**
   * 获取城市天气
   */
  static async getWeather(city: string): Promise<WeatherData> {
    try {
      // 1. 获取城市ID
      let locationId = await this.getCityId(city)

      // 2. 获取实时天气
      let currentUrl = `${this.BASE_URL}/weather/now?location=${locationId}&key=${this.API_KEY}`
      let currentData = await HttpUtil.get(currentUrl)
      let currentJson = JSON.parse(currentData)

      // 3. 获取7天预报
      let forecastUrl = `${this.BASE_URL}/weather/7d?location=${locationId}&key=${this.API_KEY}`
      let forecastData = await HttpUtil.get(forecastUrl)
      let forecastJson = JSON.parse(forecastData)

      // 4. 组装数据
      return this.parseWeatherData(city, currentJson, forecastJson)
    } catch (err) {
      throw new Error(`获取天气失败: ${err.message}`)
    }
  }

  /**
   * 获取城市ID
   */
  private static async getCityId(city: string): Promise<string> {
    let url = `${this.BASE_URL}/city/lookup?location=${city}&key=${this.API_KEY}`
    let data = await HttpUtil.get(url)
    let json = JSON.parse(data)

    if (json.code === '200' && json.location && json.location.length > 0) {
      return json.location[0].id
    }

    throw new Error('城市不存在')
  }

  /**
   * 解析天气数据
   */
  private static parseWeatherData(city: string, currentJson: any, forecastJson: any): WeatherData {
    // 当前天气
    let current = new CurrentWeather({
      city: city,
      temperature: parseInt(currentJson.now.temp),
      weather: currentJson.now.text,
      weatherIcon: this.getWeatherIcon(currentJson.now.text),
      humidity: parseInt(currentJson.now.humidity),
      windSpeed: parseInt(currentJson.now.windSpeed),
      windDirection: currentJson.now.windDir,
      updateTime: new Date().toLocaleTimeString()
    })

    // 7天预报
    let forecast: DailyForecast[] = forecastJson.daily.map((item: any) => {
      return new DailyForecast({
        date: item.fxDate,
        weekday: this.getWeekday(item.fxDate),
        tempMax: parseInt(item.tempMax),
        tempMin: parseInt(item.tempMin),
        weather: item.textDay,
        weatherIcon: this.getWeatherIcon(item.textDay)
      })
    })

    return new WeatherData({ current, forecast })
  }

  /**
   * 获取天气图标
   */
  private static getWeatherIcon(weather: string): string {
    let iconMap: Record<string, string> = {
      '晴': '☀️',
      '多云': '⛅',
      '阴': '☁️',
      '雨': '🌧️',
      '雪': '❄️',
      '雷': '⚡'
    }

    for (let key in iconMap) {
      if (weather.includes(key)) {
        return iconMap[key]
      }
    }

    return '🌤️'
  }

  /**
   * 获取星期
   */
  private static getWeekday(dateStr: string): string {
    let date = new Date(dateStr)
    let weekdays = ['日', '一', '二', '三', '四', '五', '六']
    return '周' + weekdays[date.getDay()]
  }
}
```

// ✅ 可运行代码
import { WeatherData } from '../model/WeatherModel'
import { WeatherService } from '../service/WeatherService'
import { PreferencesUtil } from '../common/utils/PreferencesUtil'

@Observed
export class WeatherViewModel {
  @Track weatherData: WeatherData = new WeatherData()
  @Track loading: boolean = false
  @Track error: string = ''
  @Track currentCity: string = '北京'

  /**
   * 加载天气数据
   */
  async loadWeather(city?: string) {
    if (city) {
      this.currentCity = city
    }

    this.loading = true
    this.error = ''

    try {
      // 从网络获取
      this.weatherData = await WeatherService.getWeather(this.currentCity)

      // 缓存数据
      await PreferencesUtil.saveWeather(this.currentCity, this.weatherData)
    } catch (err) {
      this.error = err.message

      // 加载缓存数据
      let cached = await PreferencesUtil.loadWeather(this.currentCity)
      if (cached) {
        this.weatherData = cached
        this.error += ' (显示缓存数据)'
      }
    } finally {
      this.loading = false
    }
  }

  /**
   * 刷新
   */
  async refresh() {
    await this.loadWeather()
  }
}

## 💾 步骤5: 数据持久化

**创建**: `common/utils/PreferencesUtil.ets`

```typescript
// ✅ 可运行代码
import dataPreferences from '@ohos.data.preferences'
import { WeatherData } from '../../model/WeatherModel'

export class PreferencesUtil {
  private static preferences: dataPreferences.Preferences | null = null

  /**
   * 初始化
   */
  static async init(context: Context) {
    if (!this.preferences) {
      this.preferences = await dataPreferences.getPreferences(context, 'weather_prefs')
    }
  }

  /**
   * 保存天气数据
   */
  static async saveWeather(city: string, data: WeatherData) {
    if (!this.preferences) return

    let key = `weather_${city}`
    let value = JSON.stringify(data)

    await this.preferences.put(key, value)
    await this.preferences.flush()
  }

  /**
   * 加载天气数据
   */
  static async loadWeather(city: string): Promise<WeatherData | null> {
    if (!this.preferences) return null

    let key = `weather_${city}`
    let value = await this.preferences.get(key, '')

    if (value) {
      let json = JSON.parse(value as string)
      return new WeatherData(json)
    }

    return null
  }

  /**
   * 保存当前城市
   */
  static async saveCurrentCity(city: string) {
    if (!this.preferences) return
    await this.preferences.put('current_city', city)
    await this.preferences.flush()
  }

  /**
   * 获取当前城市
   */
  static async getCurrentCity(): Promise<string> {
    if (!this.preferences) return '北京'
    return (await this.preferences.get('current_city', '北京')) as string
  }
}
```

// ✅ 可运行代码

@Entry
@Component
struct Index {

  build() {
    Column() {
      // 顶部栏
      this.buildHeader()

        this.buildError()

          Column() {
            this.buildCurrent()
            this.buildForecast()
          }

        .layoutWeight(1)
      }
    }
    .width('100%')
    .height('100%')
    .backgroundColor('#E3F2FD')

      Text(this.viewModel.currentCity)
        .fontSize(24)

        .fontSize(16)
